# 后端表结构
## 查看点单须知和陪玩价格
- 板块表（Plate）：营业板块相关的信息放进这个表里
```
struct BaseTable{
    createdAt DateTime;
    updatedAt DateTime;
    enabled bool;
    //更新者user id
    authorId int;
}

struct Plate : BaseTable{
    id int;
    name string;
    icon string;
}
```
- 长文本表（Text）：须知，协议等大段的，对用户来说只读的文本都放进这个表里
```
struct Text : BaseTable{
    id int;
    text string;
}
```
- 价格表（Price）：板块内的项目报价数据
```
// 陪玩等级
struct PlaymateLevel : BaseTable{
    id int;
    name string;
}
// 板块段位
struct PlateLevel : BaseTable{
    id int;
    name string;
}
// 加价项目
struct PlaymateAdditional : BaseTable{
    id int;
    name string;
    price int;
}
struct Price : BaseTable{
    playmateLevelId int;
    plateLevel int;
    priceType int;//0按小时 1按局
    price int;
}
```

## 陪玩列表，排行榜
- 用户表（User）：所有用户都共享的信息
```
struct User : BaseTable{
    id int;
    name string;//是否支持emoji
    birthYM DateTime;
    avatarUrl string;
    gender int;
    intro string;
    private bool;
    location string;
    phone string;
    invitedBy int;
}
```
- 陪玩表（PlaymateInfo）：陪玩专属信息
```
struct Playmate : BaseTable{
    userId int;
    playmateLevelId int;
    voidceUrl string;
    loverId int;
    online bool;
}
```
- 陪玩板块对应表（PlaymatePlate）：陪玩认证了的板块
```
struct PlaymatePlate : BaseTable{
    userId int;
    plateId int;
    authorizedDate DateTime;
}
```
- 陪玩指标(PlaymateQuality)：对陪玩的打分，每次有交易或评价完成后自动更新这个表
```
struct PlaymatePlate : BaseTable{
    userId int;
    hot int;//单量
    good int;//评价
}
```

## 订单
- 订单表（Order）
```
enum State{
    created, //付款才产生订单
    playmateRefused,
    userCancelled,
    inService,
    completed,
    judged
}

struct Order : BaseTable{
    id int;
    price int;
    count int;
    state State;
    userId int;
    playmateId int;
    plateId int;
    plateLevelId int;
    playmateLevelId int;
}

struct OrderJudge : BaseTable{
    orderId int;
    userId int;
    text string;
    rate int;
}
```

- 余额表（Charge）
```
struct Balance : BaseTable{
    userId int;
    balance int;
}

struct BalanceChange : BaseTable{
    payUserId int;
    payBalanceBefore int;
    payBalanceAfter int;
    payeeUserId int;
    payeeBalanceBefore int;
    payeeBalanceAfter int;
    desc string;
}
```
